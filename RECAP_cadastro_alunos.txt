# ATENÇÃO: dentro da f-string, use aspas DIFERENTES das externas

7) BUSCAR ALUNO PELO NOME
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

nome_busca = input("Nome a buscar: ")
encontrado = None

for aluno in alunos:
    if aluno["nome"] == nome_busca:
        encontrado = aluno
        break

if encontrado:
    print(f"Encontrado: {encontrado}")
else:
    print("Aluno não encontrado.")

# Mesma lógica da busca: encontrar antes de remover
for aluno in alunos:
    if aluno["nome"] == nome_busca:
        alunos.remove(aluno)
        print("Removido!")
        break
else:
    print("Aluno não encontrado.")

9) MÉDIA DAS NOTAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

if len(alunos) == 0:
    print("Nenhum aluno cadastrado.")
else:
    soma = 0
    for aluno in alunos:
        soma += aluno["nota"]        # soma = soma + aluno["nota"]
    media = soma / len(alunos)
    print(f"Média geral: {media:.2f}")   # .2f = 2 casas decimais

# Versão curta (usando sum + list comprehension):
# media = sum(a["nota"] for a in alunos) / len(alunos)
